# 数据流与页面跳转优化方案

## 一、现状问题分析

### 1.1 当前数据流问题
1. **页面间数据传递不连贯**: 用户在不同页面间需要重复输入相同信息
2. **状态丢失**: 页面刷新或跳转后，用户的配置状态丢失
3. **依赖关系不明确**: 用户不清楚哪些数据是必需的，哪些是可选的
4. **重复操作**: 相似的配置在不同页面需要重复设置

### 1.2 跳转体验问题
1. **缺乏上下文**: 跳转后用户不知道从哪里来，要去哪里
2. **无智能引导**: 没有基于用户当前状态的智能导航建议
3. **操作割裂**: 相关操作分散在不同页面，工作流不连贯

```typescript
// composables/useDataTransfer.ts
import { reactive, computed } from 'vue'
import { useRouter, useRoute } from 'vue-router'
import { useWorkflowStore } from '@/stores/workflow'

export function useDataTransfer() {
  const router = useRouter()
  const route = useRoute()
  const workflowStore = useWorkflowStore()

  // 传递数据到目标页面
  const transferTo = (targetRoute: string, data: Record<string, any>, options?: TransferOptions) => {
    const transferData: TransferPayload = {
      from: route.path,
      timestamp: Date.now(),
      data,
      context: {
        userAction: options?.userAction || 'navigation',
        autoFill: options?.autoFill !== false,
        preserveForm: options?.preserveForm !== false
      }
    }

    // 保存到sessionStorage供目标页面使用
    sessionStorage.setItem('dataTransfer', JSON.stringify(transferData))
    
    // 如果是模态框形式，不跳转
    if (options?.modal) {
      return openModal(targetRoute, transferData)
    }
    
    // 跳转到目标页面
    router.push({
      path: targetRoute,
      query: options?.preserveQuery ? route.query : {}
    })
  }

  // 接收传递过来的数据
  const receiveData = (): TransferPayload | null => {
    const stored = sessionStorage.getItem('dataTransfer')
    if (stored) {
      try {
        const data = JSON.parse(stored)
        // 验证数据是否过期（5分钟）
        if (Date.now() - data.timestamp < 300000) {
          return data
        }
      } catch (error) {
        console.error('Failed to parse transfer data:', error)
      }
      // 清除过期或无效数据
      sessionStorage.removeItem('dataTransfer')
    }
    return null
  }

  // 智能预填充表单
  const smartPreFill = (formData: any, transferData?: TransferPayload) => {
    if (!transferData?.context.autoFill) return formData

    const received = transferData || receiveData()
    if (!received) return formData

    // 根据数据来源页面智能映射字段
    const mappedData = mapDataBySource(received.from, received.data)
    
    // 合并数据，优先保留用户已输入的数据
    return deepMergePreservingUserInput(formData, mappedData)
  }

  // 基于页面类型的数据映射
  const mapDataBySource = (sourcePage: string, sourceData: any) => {
    const mappingRules: Record<string, (data: any) => any> = {
      '/factors': (data) => ({
        // 从因子页面来的数据映射到训练配置
        modelConfig: {
          features: data.selectedFactors || [],
          featureConfig: data.factorConfig || {}
        },
        dataConfig: {
          factors: data.selectedFactors || []
        }
      }),
      
      '/training-management': (data) => ({
        // 从训练管理页面来的数据映射到回测配置
        modelConfig: {
          modelId: data.selectedModel?.id,
          modelPath: data.selectedModel?.path,
          modelType: data.selectedModel?.type
        },
        strategyConfig: {
          // 根据模型类型推荐策略
          recommendedStrategy: getRecommendedStrategy(data.selectedModel?.type)
        }
      }),
      
      '/results': (data) => ({
        // 从结果页面来的数据映射到部署配置
        deploymentConfig: {
          strategyId: data.selectedStrategy?.id,
          strategyName: data.selectedStrategy?.name,
          performance: data.selectedStrategy?.performance,
          riskProfile: calculateRiskProfile(data.selectedStrategy?.performance)
        }
      })
    }

    const mapper = mappingRules[sourcePage]
    return mapper ? mapper(sourceData) : sourceData
  }

  return {
    transferTo,
    receiveData,
    smartPreFill
  }
}

interface TransferOptions {
  userAction?: 'navigation' | 'quick_action' | 'workflow_step'
  autoFill?: boolean
  preserveForm?: boolean
  preserveQuery?: boolean
  modal?: boolean
}

interface TransferPayload {
  from: string
  timestamp: number
  data: Record<string, any>
  context: {
    userAction: string
    autoFill: boolean
    preserveForm: boolean
  }
}
```

```typescript
// composables/useFormAutoSave.ts
import { watch, onBeforeUnmount } from 'vue'
import { useRoute } from 'vue-router'

export function useFormAutoSave(formData: any, options: AutoSaveOptions = {}) {
  const route = useRoute()
  const saveKey = options.key || `form_${route.path.replace(/\//g, '_')}`
  const saveInterval = options.interval || 30000 // 30秒自动保存

  // 从本地存储恢复数据
  const restoreFormData = () => {
    try {
      const saved = localStorage.getItem(saveKey)
      if (saved) {
        const savedData = JSON.parse(saved)
        
        // 检查数据是否过期
        if (Date.now() - savedData.timestamp < (options.expiry || 24 * 60 * 60 * 1000)) {
          return savedData.data
        } else {
          // 清除过期数据
          localStorage.removeItem(saveKey)
        }
      }
    } catch (error) {
      console.error('Failed to restore form data:', error)
    }
    return null
  }

  // 保存表单数据
  const saveFormData = () => {
    try {
      const dataToSave = {
        data: JSON.parse(JSON.stringify(formData)),
        timestamp: Date.now(),
        route: route.path
      }
      localStorage.setItem(saveKey, JSON.stringify(dataToSave))
    } catch (error) {
      console.error('Failed to save form data:', error)
    }
  }

  // 清除保存的数据
  const clearSavedData = () => {
    localStorage.removeItem(saveKey)
  }

  // 监听表单变化并自动保存
  let saveTimer: NodeJS.Timeout
  watch(
    () => formData,
    () => {
      clearTimeout(saveTimer)
      saveTimer = setTimeout(saveFormData, saveInterval)
    },
    { deep: true }
  )

  // 页面离开前保存
  onBeforeUnmount(() => {
    saveFormData()
    clearTimeout(saveTimer)
  })

  // 浏览器关闭前保存
  const handleBeforeUnload = () => {
    saveFormData()
  }
  window.addEventListener('beforeunload', handleBeforeUnload)

  onBeforeUnmount(() => {
    window.removeEventListener('beforeunload', handleBeforeUnload)
  })

  return {
    restoreFormData,
    saveFormData,
    clearSavedData
  }
}

interface AutoSaveOptions {
  key?: string
  interval?: number
  expiry?: number // 过期时间，毫秒
}
```

```typescript
// composables/useDependencyCheck.ts
export function useDependencyCheck() {
  const workflowStore = useWorkflowStore()
  
  // 检查页面访问依赖
  const checkPageDependencies = (targetPage: string): DependencyCheckResult => {
    const dependencies = getPageDependencies(targetPage)
    const missingDeps: string[] = []
    const warnings: string[] = []
    
    for (const dep of dependencies) {
      const status = checkDependency(dep)
      if (status.status === 'missing') {
        missingDeps.push(dep.name)
      } else if (status.status === 'warning') {
        warnings.push(status.message || dep.name)
      }
    }
    
    return {
      canAccess: missingDeps.length === 0,
      missingDependencies: missingDeps,
      warnings,
      suggestions: getSuggestions(targetPage, missingDeps)
    }
  }

  // 获取页面依赖配置
  const getPageDependencies = (page: string): DependencyRule[] => {
    const dependencyMap: Record<string, DependencyRule[]> = {
      '/training': [
        {
          name: 'factors',
          type: 'required',
          check: () => workflowStore.factors.available.length > 0,
          message: '需要至少1个因子才能开始训练',
          fixAction: { page: '/factors', text: '去创建因子' }
        }
      ],
      
      '/backtest': [
        {
          name: 'factors',
          type: 'required',
          check: () => workflowStore.factors.available.length > 0,
          message: '回测需要因子数据',
          fixAction: { page: '/factors', text: '去创建因子' }
        },
        {
          name: 'models',
          type: 'required',
          check: () => workflowStore.models.trained.length > 0,
          message: '回测需要训练好的模型',
          fixAction: { page: '/training', text: '去训练模型' }
        }
      ],
      
      '/deployment': [
        {
          name: 'results',
          type: 'required',
          check: () => workflowStore.strategies.completed.length > 0,
          message: '部署需要回测结果',
          fixAction: { page: '/backtest', text: '去执行回测' }
        },
        {
          name: 'good_performance',
          type: 'warning',
          check: () => workflowStore.strategies.completed.some(s => s.performance.sharpe > 1.0),
          message: '建议选择夏普比率大于1.0的策略进行部署',
          fixAction: { page: '/results', text: '查看结果详情' }
        }
      ]
    }
    
    return dependencyMap[page] || []
  }

  // 检查单个依赖
  const checkDependency = (dep: DependencyRule): DependencyStatus => {
    const isValid = dep.check()
    
    if (!isValid && dep.type === 'required') {
      return { status: 'missing', message: dep.message }
    } else if (!isValid && dep.type === 'warning') {
      return { status: 'warning', message: dep.message }
    } else {
      return { status: 'satisfied' }
    }
  }

  // 获取修复建议
  const getSuggestions = (targetPage: string, missingDeps: string[]): FixSuggestion[] => {
    const suggestions: FixSuggestion[] = []
    
    if (missingDeps.includes('factors')) {
      suggestions.push({
        title: '创建量化因子',
        description: '使用AI助手快速创建因子表达式',
        action: { page: '/factors', mode: 'ai_assistant' },
        priority: 'high'
      })
    }
    
    if (missingDeps.includes('models')) {
      suggestions.push({
        title: '训练预测模型',
        description: '使用现有因子训练机器学习模型',
        action: { page: '/training', autoFill: true },
        priority: 'high'
      })
    }
    
    return suggestions
  }

  return {
    checkPageDependencies,
    getPageDependencies
  }
}

interface DependencyRule {
  name: string
  type: 'required' | 'warning'
  check: () => boolean
  message: string
  fixAction: { page: string; text: string }
}

interface DependencyCheckResult {
  canAccess: boolean
  missingDependencies: string[]
  warnings: string[]
  suggestions: FixSuggestion[]
}

interface FixSuggestion {
  title: string
  description: string
  action: { page: string; mode?: string; autoFill?: boolean }
  priority: 'high' | 'medium' | 'low'
}
```

```vue
      actions.push(

          id: 'start_training',

          icon: 'Cpu',
          type: 'primary',
          target: '/training',
          data: { factors: workflowStore.factors.selected }
        })

      break

          type: 'primary',

  const recommendation = workflowStore.getNextStepRecommendation()
  if (!recommendation) return null
  
  return {
    title: recommendation.title,
    description: recommendation.description,
    type: recommendation.priority === 'high' ? 'warning' : 'info',
    actions: [
      {
        id: 'follow_guide',
        label: recommendation.actionText,
        primary: true,
        target: recommendation.path
      },

    action.action()
    return

    await navigateWithCheck(action.target, action.data, {
      targetName: action.label,
      userAction: 'quick_action'

  }

// 智能导航（包含依赖检查）
const navigateWithCheck = async (
  target: string, 
  data?: any, 
  options?: NavigationOptions
) => {
  const depResult = checkPageDependencies(target)
  
  if (!depResult.canAccess) {
    // 显示依赖检查对话框
    pendingNavigation.value = {
      target,
      targetName: options?.targetName || target,
      data,
      options
    }
    dependencyCheck.value = depResult
    showDependencyDialog.value = true
    return
  }
  
  // 可以正常跳转
  if (data) {
    transferTo(target, data, {
      userAction: options?.userAction || 'navigation'
    })
  } else {
    router.push(target)
  }
}

  showDependencyDialog.value = false
  
  const { page, mode, autoFill } = suggestion.action

  transferTo(page, data, {
    userAction: 'dependency_fix',
    autoFill
  })
}

  if (pendingNavigation.value) {
    const { target, data } = pendingNavigation.value
    showDependencyDialog.value = false
    
    if (data) {
      transferTo(target, data)
    } else {
      router.push(target)
    }
  }
}

// 收集上下文数据

  return {
    currentPage: route.path,
    workflowState: workflowStore.$state,
    userPreferences: workflowStore.userPreferences
  }
}

    'results': '需要完成至少1次策略回测'
```

### 3.2 页面跳转路由增强

```typescript
// router/smart-navigation.ts
import type { RouteLocationNormalized, NavigationGuardNext } from 'vue-router'
import { useWorkflowStore } from '@/stores/workflow'
import { useDependencyCheck } from '@/composables/useDependencyCheck'

// 智能导航守卫
export const smartNavigationGuard = async (
  to: RouteLocationNormalized,
  from: RouteLocationNormalized,
  next: NavigationGuardNext
) => {
  const workflowStore = useWorkflowStore()
  const { checkPageDependencies } = useDependencyCheck()
  
  // 检查页面依赖
  const depResult = checkPageDependencies(to.path)
  
  if (!depResult.canAccess) {
    // 记录被阻止的导航
    workflowStore.recordBlockedNavigation({
      from: from.path,
      to: to.path,
      reason: 'missing_dependencies',
      missingDeps: depResult.missingDependencies
    })
    
    // 如果有自动修复建议，尝试引导用户
    if (depResult.suggestions.length > 0) {
      const suggestion = depResult.suggestions[0]
      
      // 显示确认对话框
      const confirmed = await showNavigationDialog({
        title: '缺少必要条件',
        message: `访问${getPageTitle(to.path)}需要先完成前置步骤`,
        suggestions: depResult.suggestions
      })
      
      if (confirmed.action === 'redirect') {
        next(confirmed.target)
        return
      } else if (confirmed.action === 'force') {
        // 允许但记录警告
        workflowStore.recordForcedNavigation(to.path)
        next()
        return
      }
    }
    
    // 阻止导航并给出提示
    ElMessage.warning('请先完成前置步骤')
    next(false)
    return
  }
  
  // 记录成功的导航用于分析
  workflowStore.recordSuccessfulNavigation({
    from: from.path,
    to: to.path,
    timestamp: Date.now()
  })
  
  next()
}

// 数据预加载守卫
export const dataPreloadGuard = async (
  to: RouteLocationNormalized,
  from: RouteLocationNormalized,
  next: NavigationGuardNext
) => {
  // 根据目标页面预加载必要数据
  const preloadTasks = getPreloadTasks(to.path)
  
  if (preloadTasks.length > 0) {
    try {
      await Promise.all(preloadTasks.map(task => task()))
    } catch (error) {
      console.error('Data preload failed:', error)
      // 继续导航，但在页面中显示加载失败提示
    }
  }
  
  next()
}

// 获取预加载任务
const getPreloadTasks = (path: string): (() => Promise<any>)[] => {
  const tasks: (() => Promise<any>)[] = []
  
  switch (path) {
    case '/training-management':
      tasks.push(
        () => workflowStore.loadTrainingHistory(),
        () => workflowStore.loadModelLibrary()
      )
      break
      
    case '/backtest':
      tasks.push(
        () => workflowStore.loadFactorLibrary(),
        () => workflowStore.loadModelLibrary(),
        () => workflowStore.loadStrategyTemplates()
      )
      break
      
    case '/results':
      tasks.push(
        () => workflowStore.loadBacktestResults(),
        () => workflowStore.loadPerformanceMetrics()
      )
      break
  }
  
  return tasks
}

// 获取页面标题

    '/training': '模型训练',
    '/backtest': '策略回测',

}
```
