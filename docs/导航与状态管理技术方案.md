```typescript
// types/navigation.ts
interface NavigationItem {
  id: string
  title: string
  icon: string
  path: string
  badge?: number | string  // 状态提示
  children?: NavigationItem[]
  permissions?: string[]
  workflow?: {
    stage: WorkflowStage
    dependencies: string[]
    canAccess: boolean
  }
}

interface WorkflowStage {
  id: string
  name: string
  order: number
  status: 'available' | 'in-progress' | 'completed' | 'blocked'
}
```

```typescript
      return state.breadcrumbs

      this.currentPath = path
      this.updateNavigationHistory(path)
      this.updateBreadcrumbs(path)

    updateNavigationHistory(path: string) {
      // 避免重复记录相同路径
      if (this.navigationHistory[this.navigationHistory.length - 1] !== path) {
        this.navigationHistory.push(path)
        // 保持历史记录在合理长度
        if (this.navigationHistory.length > 50) {
          this.navigationHistory = this.navigationHistory.slice(-30)
        }
      }
    },

    updateBreadcrumbs(path: string) {
      const breadcrumbs: BreadcrumbItem[] = [
        { title: '首页', path: '/dashboard' }
      ]

      // 根据路径生成面包屑
      const pathSegments = path.split('/').filter(Boolean)
      let currentPath = ''
      
      for (const segment of pathSegments) {
        currentPath += `/${segment}`
        const breadcrumb = this.getBreadcrumbForPath(currentPath, segment)
        if (breadcrumb) {
          breadcrumbs.push(breadcrumb)
        }
      }

      this.breadcrumbs = breadcrumbs
    },

    getBreadcrumbForPath(path: string, segment: string): BreadcrumbItem | null {
      const pathMap: Record<string, string> = {
        '/dashboard': '仪表盘',
        '/factors': '因子开发',
        '/training': '模型训练',
        '/training-management': '训练管理',
        '/backtest': '策略回测',
        '/results': '结果分析',
        '/deployment': '策略部署',
        '/teams': '团队管理'
      }

      const title = pathMap[path]
      if (title) {
        return { title, path }
      }

      // 处理动态路由
      if (path.includes('/experiment/')) {
        return { title: '实验详情', path }
      }

      return null
    },

    addToFavorites(path: string) {
      if (!this.favoritePages.includes(path)) {
        this.favoritePages.push(path)
      }
    },

    removeFromFavorites(path: string) {
      const index = this.favoritePages.indexOf(path)
      if (index > -1) {
        this.favoritePages.splice(index, 1)
      }
    }
  }
})

interface BreadcrumbItem {
  title: string
  path: string
}
```

## 二、工作流状态管理

### 2.1 工作流Store设计

```typescript
// stores/workflow.ts
import { defineStore } from 'pinia'
import type { WorkflowState, FactorDefinition, ModelInfo, BacktestResult } from '@/types'

export const useWorkflowStore = defineStore('workflow', {
  state: (): WorkflowState => ({
    currentStage: 'dashboard',
    factors: {
      available: [],
      selected: [],
      status: 'ready'
    },
    models: {
      trained: [],
      training: [],
      selected: undefined
    },
    strategies: {
      configured: [],
      running: [],
      completed: []
    },
    deployment: {
      active: [],
      monitoring: []
    },
    dependencies: {
      dataReady: true,
      factorsReady: false,
      modelsReady: false,
      resultsReady: false
    },
    userPreferences: {
      autoSave: true,
      notifications: true,
      defaultConfigs: {}
    }
  }),

  getters: {
    workflowState: (state) => state,

    // 获取阶段状态
    getStageStatus: (state) => (stageId: string): 'available' | 'in-progress' | 'completed' | 'blocked' => {
      switch (stageId) {
        case 'factors':
          return state.factors.status === 'editing' ? 'in-progress' : 
                 state.factors.available.length > 0 ? 'completed' : 'available'
        
        case 'training':
          if (!state.dependencies.factorsReady) return 'blocked'
          return state.models.training.length > 0 ? 'in-progress' : 
                 state.models.trained.length > 0 ? 'completed' : 'available'
        
        case 'management':
          return state.models.trained.length > 0 ? 'available' : 'blocked'
        
        case 'backtest':
          if (!state.dependencies.factorsReady || !state.dependencies.modelsReady) return 'blocked'
          return state.strategies.running.length > 0 ? 'in-progress' :
                 state.strategies.completed.length > 0 ? 'completed' : 'available'
        
        case 'results':
          return state.strategies.completed.length > 0 ? 'available' : 'blocked'
        
        case 'deployment':
          const hasGoodResults = state.strategies.completed.some(s => s.performance.sharpe > 1.0)
          if (!hasGoodResults) return 'blocked'
          return state.deployment.active.length > 0 ? 'completed' : 'available'
        
        default:
          return 'available'
      }
    },

    // 获取阶段进度
    getStageProgress: (state) => (stageId: string): number => {
      switch (stageId) {
        case 'training':
          if (state.models.training.length === 0) return 0
          const totalProgress = state.models.training.reduce((sum, task) => sum + task.progress, 0)
          return Math.round(totalProgress / state.models.training.length)
        
        case 'backtest':
          if (state.strategies.running.length === 0) return 0
          const backtestProgress = state.strategies.running.reduce((sum, task) => sum + task.progress, 0)
          return Math.round(backtestProgress / state.strategies.running.length)
        
        default:
          return 0
      }
    },

    // 获取下一步推荐
    getNextStepRecommendation: (state) => (): NextStepRecommendation | null => {
      // 如果没有因子，推荐开发因子
      if (state.factors.available.length === 0) {
        return {
          title: '开始因子开发',
          description: '您还没有创建任何因子，建议先开发一些量化因子',
          actionText: '去开发因子',
          path: '/factors',
          priority: 'high'
        }
      }

      // 如果有因子但没有模型，推荐训练模型
      if (state.factors.available.length > 0 && state.models.trained.length === 0) {
        return {
          title: '训练预测模型',
          description: `您已有${state.factors.available.length}个因子，可以开始训练模型了`,
          actionText: '开始训练',
          path: '/training',
          priority: 'high'
        }
      }

      // 如果有模型但没有回测，推荐回测
      if (state.models.trained.length > 0 && state.strategies.completed.length === 0) {
        return {
          title: '执行策略回测',
          description: `您已有${state.models.trained.length}个训练好的模型，建议进行回测验证`,
          actionText: '开始回测',
          path: '/backtest',
          priority: 'high'
        }
      }

      // 如果有优秀结果但未部署，推荐部署
      const excellentResults = state.strategies.completed.filter(s => s.performance.sharpe > 1.5)
      if (excellentResults.length > 0 && state.deployment.active.length === 0) {
        return {
          title: '部署优秀策略',
          description: `发现${excellentResults.length}个表现优秀的策略，建议部署到实盘`,
          actionText: '策略部署',
          path: '/deployment',
          priority: 'medium'
        }
      }

      // 如果训练任务完成，推荐查看结果
      const recentCompleted = state.models.trained.filter(m => {
        const completedTime = new Date(m.completedAt).getTime()
        const now = Date.now()
        return now - completedTime < 3600000 // 1小时内完成的
      })
      if (recentCompleted.length > 0) {
        return {
          title: '查看训练结果',
          description: `有${recentCompleted.length}个模型刚刚训练完成，建议查看结果`,
          actionText: '查看结果',
          path: '/training-management',
          priority: 'low'
        }
      }

      return null
    }
  },

  actions: {
    // 设置当前阶段
    setActiveStage(stageId: string) {
      this.currentStage = stageId
    },

    // 更新当前路径（用于同步导航状态）
    updateCurrentPath(path: string) {
      // 根据路径推断当前阶段
      if (path.includes('/factors')) this.currentStage = 'factors'
      else if (path.includes('/training')) this.currentStage = 'training'
      else if (path.includes('/backtest')) this.currentStage = 'backtest'
      // ... 更多路径匹配
    },

    // 因子相关操作
    async addFactor(factor: FactorDefinition) {
      this.factors.available.push(factor)
      this.updateDependencies()
      
      // 自动保存
      if (this.userPreferences.autoSave) {
        await this.saveWorkflowState()
      }
    },

    async removeFactor(factorId: string) {
      const index = this.factors.available.findIndex(f => f.id === factorId)
      if (index > -1) {
        this.factors.available.splice(index, 1)
        this.updateDependencies()
      }
    },

    selectFactors(factorIds: string[]) {
      this.factors.selected = factorIds
    },

    // 模型相关操作
    async startModelTraining(config: TrainingConfig) {
      const task: TrainingTask = {
        id: generateId(),
        name: config.name,
        config,
        status: 'running',
        progress: 0,
        startedAt: new Date().toISOString()
      }
      
      this.models.training.push(task)
      
      // 调用API启动训练
      try {
        const result = await trainingApi.start(config)
        task.apiTaskId = result.taskId
      } catch (error) {
        task.status = 'failed'
        task.error = error.message
      }
    },

    async updateTrainingProgress(taskId: string, progress: number) {
      const task = this.models.training.find(t => t.id === taskId)
      if (task) {
        task.progress = progress
        
        if (progress >= 100) {
          await this.completeTraining(taskId)
        }
      }
    },

    async completeTraining(taskId: string) {
      const taskIndex = this.models.training.findIndex(t => t.id === taskId)
      if (taskIndex > -1) {
        const task = this.models.training[taskIndex]
        
        // 移到已完成列表
        const completedModel: ModelInfo = {
          id: task.id,
          name: task.name,
          type: task.config.type,
          performance: task.result?.performance || {},
          completedAt: new Date().toISOString(),
          modelPath: task.result?.modelPath
        }
        
        this.models.trained.push(completedModel)
        this.models.training.splice(taskIndex, 1)
        this.updateDependencies()
        
        // 发送通知
        if (this.userPreferences.notifications) {
          this.sendNotification(`模型 ${task.name} 训练完成`)
        }
      }
    },

    // 回测相关操作
    async startBacktest(config: BacktestConfig) {
      const task: BacktestTask = {
        id: generateId(),
        name: config.name,
        config,
        status: 'running',
        progress: 0,
        startedAt: new Date().toISOString()
      }
      
      this.strategies.running.push(task)
      
      try {
        const result = await backtestApi.start(config)
        task.apiTaskId = result.taskId
      } catch (error) {
        task.status = 'failed'
        task.error = error.message
      }
    },

    async completeBacktest(taskId: string, result: BacktestResult) {
      const taskIndex = this.strategies.running.findIndex(t => t.id === taskId)
      if (taskIndex > -1) {
        const task = this.strategies.running[taskIndex]
        
        this.strategies.completed.push({
          id: task.id,
          name: task.name,
          config: task.config,
          performance: result.performance,
          completedAt: new Date().toISOString(),
          resultPath: result.resultPath
        })
        
        this.strategies.running.splice(taskIndex, 1)
        this.updateDependencies()
      }
    },

    // 部署相关操作
    async deployStrategy(strategyId: string, config: DeploymentConfig) {
      const strategy = this.strategies.completed.find(s => s.id === strategyId)
      if (strategy) {
        const deployment: DeployedStrategy = {
          id: generateId(),
          strategyId,
          name: config.name,
          mode: config.mode,
          status: 'active',
          deployedAt: new Date().toISOString()
        }
        
        this.deployment.active.push(deployment)
        
        try {
          await deploymentApi.deploy(config)
        } catch (error) {
          deployment.status = 'failed'
          deployment.error = error.message
        }
      }
    },

    // 更新依赖状态
    updateDependencies() {
      this.dependencies.factorsReady = this.factors.available.length > 0
      this.dependencies.modelsReady = this.models.trained.length > 0
      this.dependencies.resultsReady = this.strategies.completed.length > 0
    },

    // 保存工作流状态到后端
    async saveWorkflowState() {
      try {
        await workflowApi.saveState(this.$state)
      } catch (error) {
        console.error('Failed to save workflow state:', error)
      }
    },

    // 从后端加载工作流状态
    async loadWorkflowState() {
      try {
        const state = await workflowApi.loadState()
        this.$patch(state)
      } catch (error) {
        console.error('Failed to load workflow state:', error)
      }
    },

    // 发送通知
    sendNotification(message: string) {
      // 集成通知系统
      ElNotification({
        title: '工作流通知',
        message,
        type: 'success'
      })
    }
  }
```

### 2.2 状态同步和持久化

```typescript
// composables/useWorkflowSync.ts
import { watch } from 'vue'
import { useWorkflowStore } from '@/stores/workflow'
import { useWebSocket } from '@/utils/websocket'

export function useWorkflowSync() {
  const workflowStore = useWorkflowStore()
  const { socket, isConnected } = useWebSocket()

  // 监听训练进度更新
  socket?.on('training_progress', (data: { taskId: string, progress: number }) => {
    workflowStore.updateTrainingProgress(data.taskId, data.progress)
  })

  // 监听训练完成事件
  socket?.on('training_completed', (data: { taskId: string, result: any }) => {
    workflowStore.completeTraining(data.taskId)
  })

  // 监听回测进度更新
  socket?.on('backtest_progress', (data: { taskId: string, progress: number }) => {
    workflowStore.updateBacktestProgress(data.taskId, data.progress)
  })

  // 监听回测完成事件
  socket?.on('backtest_completed', (data: { taskId: string, result: BacktestResult }) => {
    workflowStore.completeBacktest(data.taskId, data.result)
  })

  // 自动保存状态
  watch(
    () => workflowStore.$state,
    (newState) => {
      if (workflowStore.userPreferences.autoSave) {
        debounce(() => {
          workflowStore.saveWorkflowState()
        }, 2000)()
      }
    },
    { deep: true }
  )

  // 页面刷新前保存状态
  window.addEventListener('beforeunload', () => {
    workflowStore.saveWorkflowState()
  })

  return {
    isConnected
  }
}

// 防抖函数
function debounce(func: Function, wait: number) {
  let timeout: NodeJS.Timeout
  return function executedFunction(...args: any[]) {
    const later = () => {
      clearTimeout(timeout)
      func(...args)
    }
    clearTimeout(timeout)
    timeout = setTimeout(later, wait)
  }
}
```
